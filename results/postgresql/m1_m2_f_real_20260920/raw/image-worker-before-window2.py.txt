def compare(result,expected):
 if [row[0] for row in result]!=[row[0] for row in expected]:raise ValueError('image source order/identity differs')
 maximum=0.;norm_error=0.
 for (identity,value),(_,target) in zip(result,expected):
  if target is None:
   if value is not None:raise ValueError('NULL image returned a vector')
   continue
  value=np.asarray(value,dtype=np.float32);target=np.asarray(target,dtype=np.float32)
  if value.shape!=(plan.dimension,) or not np.isfinite(value).all():raise ValueError('invalid float4 vector')
  difference=float(np.max(np.abs(value-target)));maximum=max(maximum,difference)
  norm_error=max(norm_error,abs(float(np.linalg.norm(value))-1))
  if not np.allclose(value,target,rtol=c['rtol'],atol=c['atol']) or norm_error>c['norm_tolerance']:
   raise ValueError('declared CLIP numerical tolerance exceeded')
 return dict(max_abs_error=maximum,max_norm_error=norm_error,rows=len(result),non_null=sum(v is not None for _,v in result))
