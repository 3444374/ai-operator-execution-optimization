def stop_worker():

  try:

  except psutil.NoSuchProcess:pass

 except subprocess.TimeoutExpired:
  for child in children:
   try:child.terminate()
   except psutil.NoSuchProcess:pass
  _,alive=psutil.wait_procs(children,timeout=10)
  for child in alive:
   try:child.kill()
   except psutil.NoSuchProcess:pass
  worker.kill();worker.wait(10)
