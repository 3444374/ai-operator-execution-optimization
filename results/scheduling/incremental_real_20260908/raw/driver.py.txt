def save(path,value):path.write_text(redact_text(json.dumps(value,indent=2,ensure_ascii=False))+'\n')
def run(settings):
 root=Path(settings['root']);root.mkdir();repo=Path(settings['source'])
 assert subprocess.check_output(['git','-C',str(repo),'rev-parse','HEAD'],text=True).strip()==settings['source_commit'],'source_identity'
 for name,digest in settings['source_delta'].items():assert hashlib.sha256((repo/name).read_bytes()).hexdigest()==digest,'source_delta_identity'
 config=json.loads(Path(settings['config']).read_text());ledger=AttemptLedger(Path(settings['ledger']),AttemptBudget(settings['budget_id'],5));assert ledger.attempts==0,'budget_not_fresh'
 tokenizer=AutoTokenizer.from_pretrained(settings['model_root'],local_files_only=True)
 client=None;active=0;peak=0;responses={};trace=[]
 async def execute(request,endpoint):
  nonlocal client,active,peak
  assert endpoint=='model','endpoint_identity'
  if client is None:client=httpx.AsyncClient(timeout=30,trust_env=False,headers={'Content-Type':'application/json','Accept-Encoding':'identity'},limits=httpx.Limits(max_connections=2,max_keepalive_connections=2))
  ledger.reserve(hashlib.sha256(request.task.payload).hexdigest())
  active+=1;peak=max(peak,active)
  try:
   async with client.stream('POST',config['endpoint_url'],content=request.task.payload) as response:
    response.raise_for_status();buffer=bytearray()
    async for chunk in response.aiter_raw(chunk_size=4096):
     if len(buffer)+len(chunk)>request.task.max_result_bytes:raise RuntimeError('bounded_response_exceeded')
     buffer.extend(chunk)
   result=bytes(buffer);value=json.loads(result);body=json.loads(request.task.payload)
   assert value['model']==config['model_id'],'response_model'
   usage=value['usage'];assert 0<usage['completion_tokens']<=body['max_tokens'],'completion_usage'
   assert usage['prompt_tokens']==len(tokenizer.apply_chat_template(body['messages'],tokenize=True,add_generation_prompt=True,return_dict=False)),'prompt_usage'
   responses[f'{request.key.session_id}:{request.key.sequence}']={'body':body,'response':value}
   return result
  finally:active-=1
 async def finalize():
  if client is not None:await client.aclose()
 backend=BoundedAsyncBackend(execute,max_tasks=2,notify=lambda: engine.wake.notify(),finalize=finalize)
 bound=SessionLimits(2,4096,131072,2,2,3,2048,65536,64,8,120,.01)
 topology=TopologySnapshot((EndpointSnapshot('model',config['endpoint_url'],'default','0',True,0,0,0.,1.),),time.monotonic())
 engine=SessionEngine(bound,backend,SessionPolicies(StaticAdmissionController(2),RoundRobinEndpointRouter(),topology,'default'))
 def task(seq,prompt,tokens=64):
  body=build_completion_request_body(config['model_id'],[prompt],tokens,'chat_completions',temperature=0);body['stream']=False
  return OfferedTask(seq,json.dumps(body,separators=(',',':')).encode(),1,65536)
 def advance(session):
  result=session.advance(2)
  assert result.state!=State.FAILED, result.error
  trace.append({'session':session.session_id,'state':result.state.value,'delivered':[d.key.sequence for d in result.deliveries],'usage':asdict(engine.capacity.usage())})
  return result
 def collect(session,count):
  output=[];deadline=time.monotonic()+120
  while len(output)<count:
   result=advance(session);output.extend(result.deliveries)
   assert time.monotonic()<deadline,'completion_deadline'
   if len(output)<count and not result.has_immediate_work:engine.wake.wait(result.generation,.02)
  return output
 summary={'status':'incomplete','source_commit':settings['source_commit'],'source_delta':settings['source_delta'],'request_limit':5,'model_requests':0,'pg_bridge_evaluated':False,'quality_evaluated':False,'performance_evaluated':False}
 sessions=[]
 try:
  first=engine.open(SessionSpec('first','flow','fixed-chat'));sessions.append(first)
  inputs=[task(i,'Return only the word '+word+'.') for i,word in enumerate(('alpha','beta','gamma'))]
  offered=first.offer(inputs);assert offered.accepted_prefix_count==2 and offered.status=='BACKPRESSURE','accepted_prefix'
  assert not advance(first).deliveries,'submit_does_not_wait_for_model'
  assert engine.capacity.usage().active_requests==2,'two_inflight'
  deliveries=collect(first,2)
  assert engine.capacity.usage().result_bytes==131072,'unreleased_results_remain_reserved'
  assert first.offer([inputs[2]]).accepted_prefix_count==0,'release_backpressure'
  first.release([d.lease_id for d in deliveries]);assert first.offer([inputs[2]]).accepted_prefix_count==1,'accept_after_release'
  deliveries+=collect(first,1);first.release([deliveries[-1].lease_id]);first.seal();assert advance(first).state==State.FINISHED,'finish_after_seal'
  assert first.close().status=='CLOSED','first_close'
  for d in deliveries:assert json.loads(d.result)==responses[f'{d.key.session_id}:{d.key.sequence}']['response'],'delivery_bytes'
  cancelled=engine.open(SessionSpec('cancelled','flow','fixed-chat'));sessions.append(cancelled)
  cancelled.offer([task(0,'Write exactly 128 numbered short words.',128)]);advance(cancelled)
  deadline=time.monotonic()+30
  while ledger.attempts<4:
   assert time.monotonic()<deadline,'http_dispatch_deadline';engine.wake.wait(engine.wake.generation,.01)
  cancelled.cancel();report=cancelled.close();assert report.status=='CLOSED' and report.uncertain_requests==1,'retain_unpolled_responsibility'
  survivor=engine.open(SessionSpec('survivor','flow','fixed-chat'));sessions.append(survivor)
  assert engine.capacity.usage().active_requests==1,'old_reservation_survives_open'
  survivor.offer([task(0,'Return only the word delta.')]);final=collect(survivor,1)
  assert final[0].key.session_id==survivor.session_id,'old_result_not_delivered_to_new_session'
  survivor.release([final[0].lease_id]);survivor.seal();assert advance(survivor).state==State.FINISHED,'survivor_finished';survivor.close()
  deadline=time.monotonic()+120
  while engine.capacity.usage().held_tasks:
   engine.reap(8);assert not engine.error,'drain_error';assert time.monotonic()<deadline,'drain_deadline';engine.wake.wait(engine.wake.generation,.01)
  assert ledger.attempts==5 and len(responses)==5 and peak==2,'actual_request_matrix'
  summary.update(status='passed',peak_http_requests=peak,terminal_responses=len(responses),final_usage=asdict(engine.capacity.usage()),cancel_close=asdict(report),delivered_keys=[asdict(d.key) for d in deliveries+final])
 finally:
  for s in sessions:
   if s._closed is None:
    s.cancel();s.close()
  summary['model_requests']=ledger.attempts
  summary['transport_closed']=backend.close()
  if not summary['transport_closed']:summary['status']='incomplete'
  save(root/'trace.json',trace);save(root/'responses.json',responses);save(root/'summary.json',summary)
if __name__=='__main__':
 parser=argparse.ArgumentParser();parser.add_argument('--settings',type=Path,required=True);args=parser.parse_args();run(json.loads(args.settings.read_text()))
