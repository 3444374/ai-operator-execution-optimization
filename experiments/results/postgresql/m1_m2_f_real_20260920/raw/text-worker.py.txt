def new_budget(name,limit,seconds):
 if sum(item['limit'] for item in ledgers)+limit>17000:raise ValueError('text trial allocation exceeds 17000')
 if time.time()+seconds>=stop_at:raise TimeoutError('insufficient overall time for next declared stage')
 budget=AttemptBudget(name,limit);path=root/(name+'-budget.sqlite')
 ledger=CellBudgetLedger.create(path,budget,deadline_utc=time.time()+seconds-5)
 ledgers.append(dict(name=name,path=str(path),limit=limit));return budget,path,ledger

def orders(groups):
 keys=[g['id'] for g in groups]
 return [keys,keys[1:]+keys[:1],list(reversed(keys)),keys[2:]+keys[:2]]

def group(name,capacity,control='request',work=None,event='full',arm='pg'):
 value=dict(id=name,arm=arm,control=control,capacity=capacity,event_content=event)
 if work is not None:value['work_limit']=work
 return value

def stage(name,kind,split,groups,seconds,**extras):
 limit=asset['inputs'][split]['rows']*len(groups)*4
 config=dict(base,stage=kind,split=split,groups=groups,orders=orders(groups),max_posts=limit,max_seconds=seconds,
  output_root=str(root/name),budget_id=name,budget_path=str(root/(name+'-budget.sqlite')),**extras)
 check=preflight(config)
 if check['status']!='ready':raise ValueError('real stage static supply check failed')
 write_private_json(root/(name+'-preflight.json'),check)
 file=root/(name+'.json');write_private_json(file,config)
 new_budget(name,limit,seconds)
 print('START '+name+' maximum_posts='+str(limit),flush=True)
 run_campaign(file)
 result=json.loads((root/name/'comparison.json').read_text())
 phases.append(dict(stage=name,status=result['status'],actual_posts=result['actual_posts']))
 print('END '+name+' actual_posts='+str(result['actual_posts']),flush=True)
 return result,root/name/'comparison.json'

def evaluation_decision(name,groups,reason):
 file=root/(name+'-selection.json')
 write_private_json(file,dict(status='ready_for_independent_evaluation',reason=reason,
  tuning_sources=['m1-screen/comparison.json','m1-work/comparison.json'],
  evaluation_design={**{k:base[k] for k in ('resources','selection_policy','service_signature','manifest_sha256')},'groups':groups}))
 return dict(selection_source=str(file),selection_sha256=hashlib.sha256(file.read_bytes()).hexdigest())
